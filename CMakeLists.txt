cmake_minimum_required(VERSION 3.21)
project(f22 LANGUAGES CXX)
include(CTest)
add_subdirectory(EFM)

if(BUILD_TESTING)
    find_program(F22_LUA_EXECUTABLE NAMES lua5.1 lua51 lua)
    if(F22_LUA_EXECUTABLE)
        file(GLOB F22_LUA_TESTS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/tests/lua/*_test.lua")
        foreach(test_file IN LISTS F22_LUA_TESTS)
            get_filename_component(test_name "${test_file}" NAME_WE)
            add_test(NAME "lua_${test_name}" COMMAND "${F22_LUA_EXECUTABLE}" "${test_file}" "${CMAKE_SOURCE_DIR}")
        endforeach()
        file(GLOB_RECURSE F22_LUA_FILES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/*.lua")
        add_test(NAME lua_syntax COMMAND "${F22_LUA_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/lua/syntax.lua" ${F22_LUA_FILES})
    endif()
endif()
